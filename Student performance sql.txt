
-- STUDENT PERFORMANCE AND ATTENDANCE ANALYSIS
-- ORACLE SQL IMPLEMENTATION

-- QUERY 1: STUDENT-WISE PERFORMANCE ANALYSIS
-- Objective: Calculate total marks of each student
-- and arrange students in descending order.

SELECT
    STUDENT_ID,
    STUDENT_NAME,
    PYTHON + DBMS + MATHEMATICS +
    OPERATING_SYSTEMS + STATISTICS AS TOTAL_MARKS
FROM STUDENT_PERFORMANCE_DATA
ORDER BY TOTAL_MARKS DESC;


-- QUERY 2: STUDENT-WISE AVERAGE AND PERCENTAGE
-- Objective: Calculate total marks, average and percentage.

SELECT
    STUDENT_ID,
    STUDENT_NAME,
    PYTHON + DBMS + MATHEMATICS +
    OPERATING_SYSTEMS + STATISTICS AS TOTAL_MARKS,
    ROUND(
        (PYTHON + DBMS + MATHEMATICS +
        OPERATING_SYSTEMS + STATISTICS) / 5, 2
    ) AS AVERAGE_MARKS,
    ROUND(
        (PYTHON + DBMS + MATHEMATICS +
        OPERATING_SYSTEMS + STATISTICS) / 500 * 100, 2
    ) AS PERCENTAGE
FROM STUDENT_PERFORMANCE_DATA
ORDER BY TOTAL_MARKS DESC;


-- QUERY 3: ATTENDANCE PERCENTAGE ANALYSIS
-- Objective: Calculate attendance percentage of each student.

SELECT
    STUDENT_ID,
    STUDENT_NAME,
    CLASSES_HELD,
    CLASSES_ATTENDED,
    ROUND(
        (CLASSES_ATTENDED / CLASSES_HELD) * 100, 2
    ) AS ATTENDANCE_PERCENTAGE
FROM STUDENT_PERFORMANCE_DATA;


-- QUERY 4: STUDENTS WITH LOW ATTENDANCE
-- Objective: Identify students with attendance below 75%.

SELECT
    STUDENT_ID,
    STUDENT_NAME,
    CLASSES_HELD,
    CLASSES_ATTENDED,
    ROUND(
        (CLASSES_ATTENDED / CLASSES_HELD) * 100, 2
    ) AS ATTENDANCE_PERCENTAGE
FROM STUDENT_PERFORMANCE_DATA
WHERE (CLASSES_ATTENDED / CLASSES_HELD) * 100 < 75
ORDER BY ATTENDANCE_PERCENTAGE ASC;


-- QUERY 5: SUBJECT-WISE AVERAGE PERFORMANCE
-- Objective: Calculate average marks in each subject.

SELECT
    ROUND(AVG(PYTHON), 2) AS PYTHON_AVERAGE,
    ROUND(AVG(DBMS), 2) AS DBMS_AVERAGE,
    ROUND(AVG(MATHEMATICS), 2) AS MATHEMATICS_AVERAGE,
    ROUND(AVG(OPERATING_SYSTEMS), 2) AS OS_AVERAGE,
    ROUND(AVG(STATISTICS), 2) AS STATISTICS_AVERAGE
FROM STUDENT_PERFORMANCE_DATA;


-- QUERY 6: GRADE CALCULATION USING CASE
-- Objective: Assign grades based on average marks.
-- Students scoring below 40 in any subject receive F.

SELECT
    STUDENT_ID,
    STUDENT_NAME,
    ROUND(
        (PYTHON + DBMS + MATHEMATICS +
        OPERATING_SYSTEMS + STATISTICS) / 5, 2
    ) AS AVERAGE_MARKS,
    CASE
        WHEN PYTHON < 40
          OR DBMS < 40
          OR MATHEMATICS < 40
          OR OPERATING_SYSTEMS < 40
          OR STATISTICS < 40 THEN 'F'
        WHEN (PYTHON + DBMS + MATHEMATICS +
              OPERATING_SYSTEMS + STATISTICS) / 5 >= 90 THEN 'A+'
        WHEN (PYTHON + DBMS + MATHEMATICS +
              OPERATING_SYSTEMS + STATISTICS) / 5 >= 80 THEN 'A'
        WHEN (PYTHON + DBMS + MATHEMATICS +
              OPERATING_SYSTEMS + STATISTICS) / 5 >= 70 THEN 'B'
        WHEN (PYTHON + DBMS + MATHEMATICS +
              OPERATING_SYSTEMS + STATISTICS) / 5 >= 60 THEN 'C'
        WHEN (PYTHON + DBMS + MATHEMATICS +
              OPERATING_SYSTEMS + STATISTICS) / 5 >= 50 THEN 'D'
        WHEN (PYTHON + DBMS + MATHEMATICS +
              OPERATING_SYSTEMS + STATISTICS) / 5 >= 40 THEN 'E'
        ELSE 'F'
    END AS GRADE
FROM STUDENT_PERFORMANCE_DATA
ORDER BY AVERAGE_MARKS DESC;



-- QUERY 7: STUDENT RANKING USING DENSE_RANK()
-- Objective: Rank students based on total marks.

SELECT
    STUDENT_ID,
    STUDENT_NAME,
    TOTAL_MARKS,
    DENSE_RANK() OVER (
        ORDER BY TOTAL_MARKS DESC
    ) AS STUDENT_RANK
FROM (
    SELECT
        STUDENT_ID,
        STUDENT_NAME,
        PYTHON + DBMS + MATHEMATICS +
        OPERATING_SYSTEMS + STATISTICS AS TOTAL_MARKS
    FROM STUDENT_PERFORMANCE_DATA
)
ORDER BY STUDENT_RANK;



-- QUERY 8: CLASS-WISE PERFORMANCE ANALYSIS
-- Objective: Calculate average marks and student count
-- for each class and section.

SELECT
    CLASS,
    SECTION,
    COUNT(*) AS TOTAL_STUDENTS,
    ROUND(AVG(
        (PYTHON + DBMS + MATHEMATICS +
        OPERATING_SYSTEMS + STATISTICS) / 5
    ), 2) AS CLASS_AVERAGE
FROM STUDENT_PERFORMANCE_DATA
GROUP BY CLASS, SECTION
ORDER BY CLASS, SECTION;


-- 9. Grade Distribution

SELECT
    GRADE,
    COUNT(*) AS STUDENT_COUNT
FROM (
    SELECT
        CASE
            WHEN PYTHON < 40 OR DBMS < 40 OR MATHEMATICS < 40
              OR OPERATING_SYSTEMS < 40 OR STATISTICS < 40 THEN 'F'
            WHEN (PYTHON + DBMS + MATHEMATICS + OPERATING_SYSTEMS + STATISTICS) / 5 >= 90 THEN 'A+'
            WHEN (PYTHON + DBMS + MATHEMATICS + OPERATING_SYSTEMS + STATISTICS) / 5 >= 80 THEN 'A'
            WHEN (PYTHON + DBMS + MATHEMATICS + OPERATING_SYSTEMS + STATISTICS) / 5 >= 70 THEN 'B'
            WHEN (PYTHON + DBMS + MATHEMATICS + OPERATING_SYSTEMS + STATISTICS) / 5 >= 60 THEN 'C'
            WHEN (PYTHON + DBMS + MATHEMATICS + OPERATING_SYSTEMS + STATISTICS) / 5 >= 50 THEN 'D'
            WHEN (PYTHON + DBMS + MATHEMATICS + OPERATING_SYSTEMS + STATISTICS) / 5 >= 40 THEN 'E'
            ELSE 'F'
        END AS GRADE
    FROM STUDENT_PERFORMANCE_DATA
)
GROUP BY GRADE
ORDER BY CASE GRADE
    WHEN 'A+' THEN 1
    WHEN 'A' THEN 2
    WHEN 'B' THEN 3
    WHEN 'C' THEN 4
    WHEN 'D' THEN 5
    WHEN 'E' THEN 6
    ELSE 7
END;

-- 10. Create Student Result View

CREATE OR REPLACE VIEW STUDENT_RESULT_VIEW AS
SELECT
    STUDENT_ID,
    STUDENT_NAME,
    TOTAL_MARKS,
    ROUND(TOTAL_MARKS / 5, 2) AS AVERAGE_MARKS,
    ROUND((TOTAL_MARKS / 500) * 100, 2) AS PERCENTAGE,
    CASE
        WHEN PYTHON < 40 OR DBMS < 40 OR MATHEMATICS < 40
          OR OPERATING_SYSTEMS < 40 OR STATISTICS < 40 THEN 'F'
        WHEN TOTAL_MARKS / 5 >= 90 THEN 'A+'
        WHEN TOTAL_MARKS / 5 >= 80 THEN 'A'
        WHEN TOTAL_MARKS / 5 >= 70 THEN 'B'
        WHEN TOTAL_MARKS / 5 >= 60 THEN 'C'
        WHEN TOTAL_MARKS / 5 >= 50 THEN 'D'
        WHEN TOTAL_MARKS / 5 >= 40 THEN 'E'
        ELSE 'F'
    END AS GRADE
FROM (
    SELECT
        STUDENT_ID,
        STUDENT_NAME,
        PYTHON,
        DBMS,
        MATHEMATICS,
        OPERATING_SYSTEMS,
        STATISTICS,
        PYTHON + DBMS + MATHEMATICS +
        OPERATING_SYSTEMS + STATISTICS AS TOTAL_MARKS
    FROM STUDENT_PERFORMANCE_DATA
);

-- 11. Create Procedure to Display Student Grade

CREATE OR REPLACE PROCEDURE DISPLAY_STUDENT_GRADE (
    P_STUDENT_ID IN VARCHAR2
)
AS
    V_GRADE VARCHAR2(2);
BEGIN
    SELECT GRADE
    INTO V_GRADE
    FROM STUDENT_RESULT_VIEW
    WHERE STUDENT_ID = P_STUDENT_ID;

    DBMS_OUTPUT.PUT_LINE('Student Grade: ' || V_GRADE);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Student ID not found.');
END;
/
